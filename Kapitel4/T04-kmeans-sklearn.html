🧠 MiniBatchKMeans und Bewertungsmetriken

MiniBatchKMeans für große Datensätze

Statt alle Punkte in jedem Schritt zu verarbeiten, zieht MiniBatchKMeans in jedem Schritt eine zufällige Teilmenge (Mini-Batch):

KMeansMiniBatchKMeans
Zeitkomplexität pro IterationO(n·k)O(batch_size·k)
QualitätBesserLeicht schlechter (~1-2%)
SkalierungBis ~10M PunkteMilliarden Punkte möglich

Bewertungsmetriken für Clustering

MetrikBraucht Labels?Wann nutzen
Silhouette Score❌ NeinCluster-Qualität ohne Ground Truth
Davies-Bouldin Index❌ NeinNiedrig = besser
Calinski-Harabasz Index❌ NeinHoch = besser
Adjusted Rand Index (ARI)✅ JaVergleich mit Ground Truth

🛠️ Mini-Batch Update von Scratch

Python · Mini-Batch k-means Update von Scratch
import numpy as np

class MiniBatchKMeansScratch:
    """
    Mini-Batch k-means — Online-Lernvariante von k-means.
    Update-Regel: laufender Durchschnitt statt voller Neuberechnung.
    """

    def __init__(self, k=3, batch_size=100, n_iter=100, random_state=42):
        self.k = k
        self.batch_size = batch_size
        self.n_iter = n_iter
        self.rng = np.random.default_rng(random_state)

    def fit(self, X):
        X = np.array(X)
        n, d = X.shape

        # k-means++ Initialisierung
        idx = self.rng.integers(n)
        self.centroids_ = [X[idx]]

            probs = dists / dists.sum()
            self.centroids_.append(X[self.rng.choice(n, p=probs)])
        self.centroids_ = np.array(self.centroids_)

        # Counts für laufenden Durchschnitt
        counts = np.zeros(self.k)

        for _ in range(self.n_iter):
            # Mini-Batch ziehen
            batch_idx = self.rng.choice(n, self.batch_size, replace=False)
            batch = X[batch_idx]

            # Assignments für Batch
            dists = np.array([np.sum((batch - c)**2, axis=1) for c in self.centroids_])
            batch_labels = np.argmin(dists, axis=0)

            # Online-Update: laufender Durchschnitt
            for j, xi in enumerate(batch):
                c = batch_labels[j]
                counts[c] += 1
                lr = 1.0 / counts[c]
                self.centroids_[c] = (1 - lr) * self.centroids_[c] + lr * xi

        # Finale Labels
        dists = np.array([np.sum((X - c)**2, axis=1) for c in self.centroids_])
        self.labels_ = np.argmin(dists, axis=0)
        self.inertia_ = sum(
            np.sum((X[self.labels_ == k] - self.centroids_[k])**2)
            for k in range(self.k)
        )
        return self

🚀 Vollständiger Clustering-Workflow

Python · Kundensegmentierung mit KMeans
from sklearn.cluster import KMeans, DBSCAN
from sklearn.preprocessing import StandardScaler
from sklearn.metrics import (silhouette_score, davies_bouldin_score,
                              calinski_harabasz_score)
from sklearn.datasets import make_blobs
import numpy as np

# Simulierte Kundendaten: Alter, Jahreseinkommen, Ausgaben-Score
np.random.seed(42)
n = 200
means_alter = np.repeat([25, 35, 55, 45], n//4)
means_eink  = np.repeat([30, 80, 60, 40], n//4)
means_ausg  = np.repeat([20, 70, 80, 40], n//4)
X = np.column_stack([
    np.random.normal(means_alter, 5),    # Alter
    np.random.normal(means_eink, 10),    # Einkommen (k€)
    np.random.normal(means_ausg, 8),     # Ausgaben 0-100
])

scaler = StandardScaler()
X_s = scaler.fit_transform(X)

# Optimales k finden
print("k-Analyse:")
print(f"{'k':>4} | {'Inertia':>10} | {'Silhouette':>12} | {'DB-Index':>10}")
print("-" * 45)
for k in range(2, 7):
    km = KMeans(n_clusters=k, n_init=10, random_state=42)
    labels = km.fit_predict(X_s)
    sil = silhouette_score(X_s, labels)
    db  = davies_bouldin_score(X_s, labels)
    print(f"{k:>4} | {km.inertia_:>10.2f} | {sil:>12.4f} | {db:>10.4f}")

# Finales Modell mit k=4
km = KMeans(n_clusters=4, n_init=10, random_state=42)
labels = km.fit_predict(X_s)

# Cluster-Profile: Rücktransformieren für Interpretation
centroids_orig = scaler.inverse_transform(km.cluster_centers_)
feature_names = ['Alter', 'Einkommen(k€)', 'Ausgaben']

print("\n--- Cluster-Profile ---")
print(f"{'Cluster':<10} {'Größe':>6} {'Alter':>8} {'Einkommen':>11} {'Ausgaben':>9}")
print("-" * 48)
for i in range(4):
    size = np.sum(labels == i)
    c = centroids_orig[i]
    print(f"Cluster {i:<3} {size:>6}   {c[0]:>7.1f}   {c[1]:>9.1f}   {c[2]:>8.1f}")

# Clustering als Preprocessing: Labels als Feature hinzufügen
X_with_clusters = np.column_stack([X_s, labels])
print(f"\nX_with_clusters shape: {X_with_clusters.shape}")
