1 Scaler & Train/Test-Split

Z-Score-Standardisierung und ein einfacher Split ohne sklearn.

Python 
def standardize(X_train, X_test):
    mu = X_train.mean(axis=0)
    sigma = X_train.std(axis=0)
    X_train_scaled = (X_train - mu) / sigma
    X_test_scaled = (X_test - mu) / sigma   # gleiche mu/sigma wie Train!
    return X_train_scaled, X_test_scaled
Python 
def train_test_split_scratch(X, y, test_size=0.2, seed=42):
    np.random.seed(seed)
    n = len(X)
    idx = np.random.permutation(n)
    split = int(n * (1 - test_size))
    train_idx, test_idx = idx[:split], idx[split:]
    return X[train_idx], X[test_idx], y[train_idx], y[test_idx]

2 Gini, Entropie & Information Gain

Splitkriterien für Entscheidungsbäume.

Python 
def gini(y):
    _, counts = np.unique(y, return_counts=True)
    p = counts / len(y)
    return 1 - np.sum(p**2)
Python 
def entropy(y):
    _, counts = np.unique(y, return_counts=True)
    p = counts / len(y)
    return -np.sum(p * np.log2(p))
Python 
def information_gain(y, y_left, y_right, criterion=gini):
    n, n_l, n_r = len(y), len(y_left), len(y_right)
    weighted_child = (n_l/n)*criterion(y_left) + (n_r/n)*criterion(y_right)
    return criterion(y) - weighted_child

3 Entscheidungsbaum: Besten Split finden

Iteriert über alle Features und alle Schwellenwerte, wählt den Split mit dem höchsten Information Gain.

Python 
def best_split(X, y):
    best_gain, best_feature, best_threshold = -1, None, None
    for feature_idx in range(X.shape[1]):
        thresholds = np.unique(X[:, feature_idx])
        for t in thresholds:
            left_mask = X[:, feature_idx] <= t
            right_mask = ~left_mask
            if left_mask.sum() == 0 or right_mask.sum() == 0:
                continue
            gain = information_gain(y, y[left_mask], y[right_mask])
            if gain > best_gain:
                best_gain, best_feature, best_threshold = gain, feature_idx, t
    return best_feature, best_threshold, best_gain

4 k-Means-Algorithmus

Init → Zuweisung → Update, bis Konvergenz oder max_iter.

Python 
def kmeans_scratch(X, k, max_iter=100, seed=42):
    np.random.seed(seed)
    centroids = X[np.random.choice(len(X), k, replace=False)]

    for _ in range(max_iter):
        # 1. Zuweisung: Distanz jeder Probe zu jedem Centroid
        dists = np.linalg.norm(X[:, None, :] - centroids[None, :, :], axis=2)
        labels = np.argmin(dists, axis=1)

        # 2. Update: neue Centroids = Mittelwert der zugewiesenen Punkte
        new_centroids = np.array([
            X[labels == i].mean(axis=0) if np.any(labels == i) else centroids[i]
            for i in range(k)
        ])

        if np.allclose(centroids, new_centroids):
            break
        centroids = new_centroids

    wcss = np.sum((X - centroids[labels])**2)
    return labels, centroids, wcss

5 Bagging (Bootstrap Aggregating)

Trainiert mehrere Modelle auf Bootstrap-Stichproben und kombiniert per Mehrheitsentscheid (Klassifikation) bzw. Mittelwert (Regression).

Python 
def bagging_fit(X, y, n_estimators=10, base_estimator_fn=None, seed=42):
    np.random.seed(seed)
    n = len(X)
    models = []
    for _ in range(n_estimators):
        idx = np.random.choice(n, n, replace=True)   # Bootstrap: mit Zurücklegen
        model = base_estimator_fn()
        model.fit(X[idx], y[idx])
        models.append(model)
    return models
Python 
def bagging_predict(models, X):
    preds = np.array([m.predict(X) for m in models])  # shape (n_estimators, n_samples)
    # Mehrheitsentscheid pro Sample (Klassifikation)
    return np.array([np.bincount(preds[:, i]).argmax() for i in range(X.shape[0])])

6 EM-Algorithmus (Gaussian Mixture)

E-Step: Verantwortlichkeiten (Soft-Zuordnung) berechnen. M-Step: Parameter (μ, Σ, π) anhand der Verantwortlichkeiten neu schätzen.

Python 
from scipy.stats import multivariate_normal

def e_step(X, means, covariances, weights):
    k = len(weights)
    resp = np.zeros((len(X), k))
    for j in range(k):
        resp[:, j] = weights[j] * multivariate_normal.pdf(X, means[j], covariances[j])
    resp /= resp.sum(axis=1, keepdims=True)   # normieren -> Wahrscheinlichkeiten je Sample
    return resp
Python 
def m_step(X, resp):
    Nk = resp.sum(axis=0)                      # "effektive" Anzahl Punkte pro Cluster
    weights = Nk / len(X)
    means = (resp.T @ X) / Nk[:, None]
    covariances = []
    for j in range(len(Nk)):
        diff = X - means[j]
        cov = (resp[:, j, None, None] * np.einsum('ni,nj->nij', diff, diff)).sum(axis=0) / Nk[j]
        covariances.append(cov)
    return means, np.array(covariances), weights

7 Confusion-Matrix-Metriken

Accuracy, Precision, Recall, F1 direkt aus TP/FP/FN/TN — siehe auch den Live-Rechner im Cheat-Sheet.

Python 
def confusion_counts(y_true, y_pred):
    tp = np.sum((y_true == 1) & (y_pred == 1))
    fp = np.sum((y_true == 0) & (y_pred == 1))
    fn = np.sum((y_true == 1) & (y_pred == 0))
    tn = np.sum((y_true == 0) & (y_pred == 0))
    return tp, fp, fn, tn
Python 
def metrics_from_counts(tp, fp, fn, tn):
    accuracy  = (tp + tn) / (tp + fp + fn + tn)
    precision = tp / (tp + fp) if (tp + fp) > 0 else 0.0
    recall    = tp / (tp + fn) if (tp + fn) > 0 else 0.0
    specificity = tn / (tn + fp) if (tn + fp) > 0 else 0.0
    f1 = (2 * precision * recall / (precision + recall)
          if (precision + recall) > 0 else 0.0)
    return {"accuracy": accuracy, "precision": precision,
            "recall": recall, "specificity": specificity, "f1": f1}

8 PCA via Eigenzerlegung

Zentrieren → Kovarianzmatrix → Eigenzerlegung → absteigend sortieren → projizieren.

Python 
def pca_fit(X, n_components):
    mean = X.mean(axis=0)
    X_centered = X - mean

    # Kovarianzmatrix (Features x Features)
    cov = np.cov(X_centered, rowvar=False)

    # Eigenzerlegung -- Sigma ist symmetrisch -> eigh (aufsteigend sortiert)
    eigvals, eigvecs = np.linalg.eigh(cov)

    # absteigend sortieren: größter Eigenwert = PC1
    order = np.argsort(eigvals)[::-1]
    eigvals = eigvals[order]
    eigvecs = eigvecs[:, order]

    components = eigvecs[:, :n_components].T          # (n_components, n_features)
    explained_variance = eigvals[:n_components]
    explained_variance_ratio = explained_variance / eigvals.sum()

    return mean, components, explained_variance_ratio
Python 
def pca_transform(X, mean, components):
    return (X - mean) @ components.T

def pca_inverse_transform(X_proj, mean, components):
    return X_proj @ components + mean

9 Gradient Boosting

Startwert = Mittelwert von y → Residuen berechnen → flachen Baum auf Residuen fitten → additiv mit learning_rate aktualisieren → wiederholen.

Python 
from sklearn.tree import DecisionTreeRegressor

def gb_fit(X, y, n_estimators=20, learning_rate=0.5, max_depth=1):
    F0 = np.mean(y)
    models = []
    pred = np.full(len(y), F0)
    for _ in range(n_estimators):
        residuals = y - pred                # Pseudo-Gradient (squared_error)
        tree = DecisionTreeRegressor(max_depth=max_depth)
        tree.fit(X, residuals)
        pred = pred + learning_rate * tree.predict(X)
        models.append(tree)
    return F0, models
Python 
def gb_predict(F0, models, X, learning_rate=0.5):
    pred = np.full(X.shape[0], F0)
    for tree in models:
        pred = pred + learning_rate * tree.predict(X)
    return pred
