1 Matplotlib Grundlagen

Setup

import matplotlib.pyplot as plt
import numpy as np

    

    

      

        
plt.plot(x, y)

        
Liniendiagramm — z.B. Lernkurve, Verlustfunktion

      

      
x = np.linspace(0, 10, 100)
plt.plot(x, np.sin(x))
plt.show()

      
# zeigt Liniendiagramm (Sinuskurve)

    

    

      

        
plt.scatter(x, y)

        
Streudiagramm — z.B. Datenpunkte, Cluster-Ergebnisse

      

      
plt.scatter(X[:, 0], X[:, 1], 
            c=labels, cmap='viridis')
plt.show()

      
# zeigt Punktwolke, Farbe = Cluster-Label

    

    

      

        
plt.bar(x, height) / plt.hist(data)

        
Balkendiagramm bzw. Histogramm — Verteilungen

      

      
plt.hist(data, bins=20)
plt.show()

      
# zeigt Häufigkeitsverteilung in 20 Bins

    

    

      

        
plt.title() / plt.xlabel() / plt.ylabel()

        
Titel und Achsenbeschriftung setzen

      

      
plt.title("Lernkurve")
plt.xlabel("Trainingsgröße")
plt.ylabel("Score")

      
# beschriftet Titel, x- und y-Achse

    

    

      

        
plt.legend()

        
Legende anzeigen — benötigt label= in plot()/scatter()

      

      
plt.plot(x, y1, label="Train")
plt.plot(x, y2, label="Validation")
plt.legend()

      
# zeigt Legende mit 
            "Train" und "Validation"

    

    

      

        
plt.subplots(nrows, ncols)

        
Mehrere Plots in einer Figure anordnen — gibt fig, axes zurück

      

      
fig, axes = plt.subplots(1, 2, figsize=(10, 4))
axes[0].plot(x, np.sin(x))
axes[0].set_title("Sinus")
axes[1].scatter(X[:, 0], X[:, 1])
axes[1].set_title("Daten")
plt.tight_layout()
plt.show()

      
# axes ist ein Array von Axes-Objekten — axes.shape == (2,) bei 1 Zeile, (nrows, ncols) bei mehreren Zeilen

    

    

      

        
plt.figure(figsize=(w,h))

        
Größe der Figure in Zoll festlegen

      

      
plt.figure(figsize=(8, 5))
plt.plot(x, y)

      
# Figure mit Breite 8", Höhe 5"

    

    

      

        
plt.savefig(path)

        
Plot als Datei speichern — vor plt.show() aufrufen

      

      
plt.plot(x, y)
plt.savefig("plot.png", dpi=150, 
            bbox_inches='tight')
plt.show()

      
# speichert plot.png mit 150 dpi, 
            ohne überschüssigen Rand

    

    

      

        
plt.axhline() / plt.axvline()

        
Horizontale/vertikale Referenzlinie — z.B. Threshold, Mittelwert

      

      
plt.axhline(y=0.5, color='r', 
            linestyle='--', 
            label='Threshold')

      
# gestrichelte rote Linie bei y=0.5

    

    

      

        
plt.colorbar()

        
Farbskala neben Scatter/Heatmap anzeigen

      

      
sc = plt.scatter(X[:, 0], X[:, 1], c=y)
plt.colorbar(sc, label="Klasse")

      
# zeigt Farbskala mit Beschriftung "Klasse"

    

  

  
      

        Matplotlib-Plots erzeugen keinen Text-Output — in Prüfungs-Notebooks reicht meist der korrekte Code (Achsenbeschriftung, Legende, Titel). Wichtig: plt.show() am Ende, sonst bleiben Plots evtl. unsichtbar.

4 Seaborn Plots

        

    

      

        
Setup

      

      
import seaborn as sns

      

    

    

      

        
sns.scatterplot(data=df, x=, y=, hue=)

        
Streudiagramm mit automatischer Farbgebung nach Kategorie

      

      
sns.scatterplot(data=df, x='sepal_length', 
        y='petal_width', hue='species')
plt.show()

      
# Punktwolke, Farbe = species, inkl. Legende

    

    

      

        
sns.heatmap(matrix, annot=True)

        
Matrix als Farbraster — z.B. Korrelationen, Confusion Matrix

      

      
sns.heatmap(df.corr(numeric_only=True), 
            annot=True, cmap='coolwarm')
plt.show()

      
# Korrelationsmatrix mit 
            Zahlenwerten in jeder Zelle

    

    

      

        
sns.boxplot(data=df, x=, y=)

        
Verteilung pro Kategorie — Median, Quartile, Ausreißer

      

      
sns.boxplot(data=df, x='species', y='petal_width')
plt.show()

      
# Box-Plot pro species mit 
            Median, IQR, Ausreißer-Punkten

    

    

      

        
sns.histplot(data=df, x=, hue=, kde=True)

        
Histogramm mit optionaler Dichteschätzung (KDE)

      

      
sns.histplot(data=df, x='petal_width', hue='species', kde=True)
plt.show()

      
# Histogramm + glatte Dichtekurve, 
            pro species eingefärbt

    

    

      

        
sns.pairplot(df, hue='col')

        
Alle Feature-Paare als Scatter-Matrix — schneller Überblick über Zusammenhänge

      

      
sns.pairplot(df, hue='species')
plt.show()

      
# Gitter aus Scatterplots für 
jedes Feature-Paar + Histogramme auf der Diagonale

    

    

      

        
sns.lineplot(data=df, x=, y=)

        
Liniendiagramm mit automatischem Konfidenzband bei mehreren Werten

      

      
sns.lineplot(data=df, x='epoch', y='loss')
plt.show()

      
# Linienverlauf, Schattierung = Konfidenzintervall

    

  

  
      

        Seaborn baut auf Matplotlib auf — plt.show(), plt.title(), plt.xlabel() etc. funktionieren genauso. Mit data=df und Spaltennamen als x=/y=/hue= entfällt manuelles Indexieren.
